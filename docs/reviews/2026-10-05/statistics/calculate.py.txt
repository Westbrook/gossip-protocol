def probabilities(n,pw,pl,alpha,margin=.1):
 lower=[stats._cp_lower(k,n,alpha/4) for k in range(n+1)]
 upper=[1-lower[n-k] for k in range(n+1)]
 flags={k:[] for k in ['worthwhile_benefit','exclude_worthwhile_benefit','practical_equivalence','directional_harm','worthwhile_harm','unresolved_benefit_equivalence_harm']}
 for w,wm in enumerate(stats.binomial_masses(n,pw)):
  if not wm:continue
  subtotal={k:[] for k in flags}
  for l,lm in enumerate(stats.binomial_masses(n-w,pl/(1-pw))):
   lo=lower[w]-upper[l];hi=upper[w]-lower[l]
   a=lo>margin;b=hi<margin;c=lo>-margin and hi<margin;d=hi<0;e=hi<-margin
   for key,yes in zip(flags,[a,b,c,d,e,not(a or c or e)]):
    if yes:subtotal[key].append(lm)
  for key in flags:flags[key].append(wm*math.fsum(subtotal[key]))
 ans={k:math.fsum(v) for k,v in flags.items()}
 existing=stats.interval_decision_power(n,pw,pl,margin,alpha)
 assert abs(ans['worthwhile_benefit']-existing['meaningful_benefit'])<1e-10
 assert abs(ans['exclude_worthwhile_benefit']-existing['rules_out_meaningful_benefit'])<1e-10
 assert abs(ans['worthwhile_benefit']+ans['practical_equivalence']+ans['worthwhile_harm']+ans['unresolved_benefit_equivalence_harm']-1)<1e-10
 return ans
